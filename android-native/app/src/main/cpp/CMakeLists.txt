cmake_minimum_required(VERSION 3.22.1)
project(onevoice_audio LANGUAGES C CXX)
file(GLOB SHINE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/shine/src/lib/*.c")
if(NOT SHINE_SOURCE)
    message(FATAL_ERROR "Missing Shine source. The GitHub build workflow downloads the permissively licensed upstream library.")
endif()
add_library(onevoice-mp3 SHARED mp3_bridge.cpp ${SHINE_SOURCE})
target_include_directories(onevoice-mp3 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/shine/src/lib")
target_compile_options(onevoice-mp3 PRIVATE -O2)
target_link_libraries(onevoice-mp3 m log)
